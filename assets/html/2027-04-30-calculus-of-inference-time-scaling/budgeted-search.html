Budgeted search under controlled assumptions

Simulation, not LLM benchmark: six-step task, independently successful steps with probability s; a bad prefix remains bad. Proxy score = q for a valid prefix or 1−q otherwise, plus uniform noise U(0,.8). q is score separation, not measured calibration. Each generated step costs one unit; verifier scoring, parallel hardware and KV costs are excluded. The beam branches twice per retained prefix, uses width N/2 and never exceeds 6N generated steps. Sequential revision is a separate assumed transition model (correct stays correct with .85, incorrect repairs with min(.95,p+.08)); results depend on these assumptions. It is not MCTS. 1,200 trials per condition; 36 conditions, 43,200 condition-trials, four strategies. Same seed within a condition across budgets. Entries are mean ± one binomial standard error in percentage points. Oracle sampling is an upper bound for reranking the same candidates, not for all search procedures.

function experiment(){
 const N=1200,D=6,tiers=[.50,.75,.90],qualities=[.60,.85],budgets=[2,4,8,16,32,64];
 function rng(seed){let a=seed>>>0;return ()=>{a=(a+0x6D2B79F5)>>>0;let t=a;t=Math.imul(t^(t>>>15),t|1);t^=t+Math.imul(t^(t>>>7),t|61);return ((t^(t>>>14))>>>0)/4294967296}}
 function choose(xs,r,q){let best=-1,win=false;for(const x of xs){let score=(x?q:1-q)+(.8*r());if(score>best){best=score;win=x}}return +win;}
 function independent(r,s){let x=true;for(let d=0;d<D;d++){let good=r()<s;x=x&&good}return x}
 function beam(r,s,q,n){let xs=[true],w=Math.max(1,n/2);let used=0;for(let d=0;d<D;d++){let next=[];for(const x of xs)for(let j=0;j<2;j++){let y=x&&r()<s;used++;next.push([y,(y?q:1-q)+.8*r()])}next.sort((a,b)=>b[1]-a[1]);xs=next.slice(0,w).map(x=>x[0])}if(used>D*n)throw Error("budget");return [+xs[0],used]}
 const rows=[];for(let ti=0;ti<tiers.length;ti++)for(let qi=0;qi<qualities.length;qi++)for(const n of budgets){let s=tiers[ti],q=qualities[qi],p=s**D,arr=[0,0,0,0],diff=[],cost=0;for(let i=0;i<N;i++){let r=rng(940000+ti*100000+qi*10000+i),xs=Array.from({length:n},()=>independent(r,s));const y=[+xs.some(Boolean),choose(xs,r,q)];let br=beam(rng(1940000+ti*100000+qi*10000+i),s,q,n);y.push(br[0]);cost+=br[1];let rr=rng(2940000+ti*100000+qi*10000+i);let prev=rr()<p,seq=[prev];for(let t=1;t<n;t++){prev=rr()<(prev?.85:Math.min(.95,p+.08));seq.push(prev)}y.push(choose(seq,rr,q));y.forEach((v,j)=>arr[j]+=v);diff.push(y[2]-y[1]);}let m=diff.reduce((a,b)=>a+b,0)/N,sd=Math.sqrt(diff.reduce((a,b)=>a+(b-m)**2,0)/(N-1)/N);rows.push({s,q,n,p,accuracy:arr.map(v=>v/N),se:arr.map(v=>Math.sqrt((v/N)*(1-v/N)/N)),beamCost:cost/N,delta:m,half:2*sd,oracle:1-(1-p)**n})}return rows;
}
