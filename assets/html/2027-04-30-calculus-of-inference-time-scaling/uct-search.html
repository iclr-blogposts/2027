Finite-tree search: sampling, beam and UCT

Actual executed synthetic tree experiment; not a benchmark of LLMs. Each independent task is a fixed full binary tree of depth 7. Root valid; each child of a valid prefix remains valid independently with probability p. Invalid branches are absorbing. Every prefix/leaf score is fixed at .5 ± separation + U(−.3,.3). All algorithms use the same tree and terminal score for final selection. Beam can additionally score intermediate prefixes, and UCT backs up terminal proxy scores (exploration coefficient √2). Each traversed/generated edge costs one unit even if cached or repeated. Beam uses at most 7N edges, sampling and UCT exactly 7N. No wall-clock or FLOP measurement. 800 tasks per condition; 20 conditions, 16,000 task-condition evaluations and 48,000 algorithm evaluations. The same 800 task trees recur across budgets within each p/separation condition (3,200 distinct trees). Different methods use paired task instances. Report unique leaves, generated edges and verifier calls separately.

function runTreeStudy(){
 const H=7,N=800,tiers=[.65,.85],separations=[.1,.3],budgets=[4,8,16,32,64],U=Math.sqrt(2);
 function rng(seed){let a=seed>>>0;return ()=>{a=(a+0x6D2B79F5)>>>0;let t=a;t=Math.imul(t^(t>>>15),t|1);t^=t+Math.imul(t^(t>>>7),t|61);return ((t^(t>>>14))>>>0)/4294967296}}
 function tree(r,p,sep){const size=2**(H+1),good=new Uint8Array(size),v=new Float64Array(size);good[1]=1;for(let i=2;i<size;i++){good[i]=+(good[i>>1]&&r()<p);v[i]=.5+(good[i]?sep:-sep)+.6*(r()-.5)}return {good,v}}
 function solve(env,r,mode,n){const {good,v}=env;let best=-1,chosen=0,found=false,seen=new Set(),edges=0,prefixScores=0,terminalScores=0;
 function visit(i){terminalScores++;seen.add(i);if(good[i])found=true;if(v[i]>best){best=v[i];chosen=i}}
 if(mode==='beam'){let cur=[1],w=Math.max(1,Math.floor(n/2));for(let d=0;d<H;d++){let next=[];for(const i of cur){next.push(i*2,i*2+1);edges+=2}if(d===H-1){for(const i of next)visit(i)}else{prefixScores+=next.length;next.sort((a,b)=>v[b]-v[a]||a-b);cur=next.slice(0,w)}}}
 else if(mode==='sample'){for(let j=0;j<n;j++){let i=1;for(let d=0;d<H;d++){i=i*2+ +(r()<.5);edges++}visit(i)}}
 else {const count=new Uint32Array(good.length),sum=new Float64Array(good.length);for(let j=0;j<n;j++){let i=1,path=[1],expanded=false;for(let d=0;d<H;d++){let a=i*2,b=a+1;if(expanded){i=r()<.5?a:b}else if(!count[a]||!count[b]){i=(!count[a]&&!count[b])?(r()<.5?a:b):!count[a]?a:b;expanded=true;path.push(i)}else{let sa=sum[a]/count[a]+U*Math.sqrt(Math.log(count[i])/count[a]),sb=sum[b]/count[b]+U*Math.sqrt(Math.log(count[i])/count[b]);i=sa===sb?(r()<.5?a:b):sa>sb?a:b;path.push(i)}edges++}visit(i);for(const a of path){count[a]++;sum[a]+=v[i]}}}
 if(edges>n*H)throw Error('budget exceeded');return [+!!good[chosen],+found,seen.size,edges,prefixScores,terminalScores]}
 const rows=[];for(let ti=0;ti<tiers.length;ti++)for(let si=0;si<separations.length;si++)for(const n of budgets){const vals=Array.from({length:3},()=>Array(6).fill(0)),paired=[];let solvable=0;for(let e=0;e<N;e++){const seed=381000+ti*100000+si*10000+e,env=tree(rng(seed),tiers[ti],separations[si]);solvable+=+(env.good.slice(2**H).some(Boolean));const o=['sample','beam','uct'].map((m,j)=>solve(env,rng(seed+700000+j*1000000),m,n));o.forEach((a,j)=>a.forEach((x,k)=>vals[j][k]+=x));paired.push(o[2][0]-o[0][0])}const avg=vals.map(a=>a.map(x=>x/N)),diff=paired.reduce((s,x)=>s+x,0)/N,se=Math.sqrt(paired.reduce((s,x)=>s+(x-diff)**2,0)/(N-1)/N);if(avg.some(a=>a[0]>a[1]+1e-10||a[1]>solvable/N+1e-10))throw Error('coverage invariant');rows.push({p:tiers[ti],sep:separations[si],n,solvable:solvable/N,avg,diff,half:2*se})}return rows
}

Coverage measures whether any visited terminal is valid; accuracy measures the selected terminal. A gap is a selection error. Different verifier-call counts mean a matched generated-edge budget is not a matched total-compute budget. This UCT experiment does not infer anything about the private implementation of commercial reasoning systems.
