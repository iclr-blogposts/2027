Rare verifier errors under selection pressure

Executed synthetic experiment, not a benchmark of a learned verifier. Correctness probability .25; correct scores .6+U(0,.5); incorrect scores .2+U(0,.5), except a fraction h of incorrect candidates receives 1.4+U(0,.2). The second verifier has no rare tail by construction but shares the same correctness labels; averaging is not guaranteed to eliminate false positives. Each condition has 3,000 independent candidate streams; nested budgets reuse candidates. Total 9,000 streams and 2,304,000 candidate draws. Proxy score is not a calibrated probability. The exact probability of no catastrophic false positive and at least one correct sample is (1−.75h)^n − [.75(1−h)]^n. It is an upper bound on correct selection because ordinary incorrect scores may still beat correct scores.

function simulate(){
 function rng(seed){let a=seed>>>0;return ()=>{a=(a+0x6D2B79F5)>>>0;let t=a;t=Math.imul(t^(t>>>15),t|1);t^=t+Math.imul(t^(t>>>7),t|61);return ((t^(t>>>14))>>>0)/4294967296}}
 const N=3000,budgets=[1,2,4,8,16,32,64,128,256],rates=[0,.005,.02],rows=[];
 for(let j=0;j<rates.length;j++){let h=rates[j],sum=budgets.map(()=>[0,0,0,0,0]);for(let i=0;i<N;i++){let r=rng(800000+j*100000+i),best=-Infinity,selected=false,oracle=false,rare=false;let best2=-Infinity,sel2=false;for(let n=1;n<=256;n++){let y=r()<.25,tail=!y&&r()<h,base=(y?.6:.2),s=tail?1.4+.2*r():base+.5*r(),s2=base+.5*r();oracle=oracle||y;if(s>best){best=s;selected=y;rare=tail}if((s+s2)/2>best2){best2=(s+s2)/2;sel2=y}let k=budgets.indexOf(n);if(k>=0){sum[k][0]+=+selected;sum[k][1]+=+oracle;sum[k][2]+=+sel2;sum[k][3]+=best;sum[k][4]+=+rare}}}rows.push({h,values:budgets.map((n,i)=>({n,accuracy:sum[i][0]/N,oracle:sum[i][1]/N,ensemble:sum[i][2]/N,proxy:sum[i][3]/N,rare:sum[i][4]/N,exact:(1-.75*h)**n-.75**n*(1-h)**n}))})}return rows;
}
