Calibration is not ranking

Executed synthetic study, not a real verifier benchmark. Draw latent p uniformly in [.02,.98], then Y ~ Bernoulli(sigmoid(logit(p)+shift)). Raw score = sigmoid(2.5 logit(p)). Choose temperature by minimum validation log loss on 76 temperatures from .5 to 4.25. Freeze it for independent IID and shifted test sets. 16,000 examples in each of three splits (48,000 total). Shifted labels use intercept −1.2; raw score distribution is unchanged in expectation. Selected temperature: 2.60. Equal-width ECE, 10 bins. AUROC is computed on the untransformed logit and is invariant to a strictly increasing temperature map (apart from numerical ties). No claim of calibration under selection or distribution shift.

Reproduce in Node.js
function run(){
 const bins=10, N=16000, seed=13579;
 function rng(a){return ()=>{a=(a+0x6D2B79F5)>>>0;let t=a;t=Math.imul(t^(t>>>15),t|1);t^=t+Math.imul(t^(t>>>7),t|61);return ((t^(t>>>14))>>>0)/4294967296}}
 const sigmoid=x=>1/(1+Math.exp(-x)),logit=p=>Math.log(p/(1-p));
 function data(n,seed,shift){let r=rng(seed),out=[];for(let i=0;i<n;i++){let p=.02+.96*r(),y=+(r()<sigmoid(logit(p)+shift));out.push({x:2.5*logit(p),y})}return out}
 function metrics(data,t){let b=0,l=0,bb=Array.from({length:bins},()=>[0,0,0]);for(const {x,y} of data){let p=sigmoid(x/t);b+=(p-y)**2;l-=y*Math.log(Math.max(p,1e-15))+(1-y)*Math.log(Math.max(1-p,1e-15));let g=bb[Math.min(bins-1,Math.floor(p*bins))];g[0]++;g[1]+=p;g[2]+=y;}let ece=bb.reduce((a,[n,p,y])=>a+Math.abs(p-y),0)/data.length;return {brier:b/data.length,nll:l/data.length,ece,bins:bb.map(([n,p,y])=>[n,n?p/n:0,n?y/n:0])}}
 function auc(data){let xs=data.slice().sort((a,b)=>a.x-b.x),sum=0,pos=0,neg=0;for(const a of xs){if(a.y){sum+=neg;pos++}else neg++}return sum/(pos*neg)}
 const val=data(N,seed,0),test=data(N,seed+1,0),shift=data(N,seed+2,-1.2);
 let grid=Array.from({length:76},(_,i)=>.5+i*.05);let temp=grid.reduce((a,b)=>metrics(val,b).nll<metrics(val,a).nll?b:a,1);
 return {N,temp,rows:[{name:"IID test",auc:auc(test),raw:metrics(test,1),scaled:metrics(test,temp)},{name:"Shifted test",auc:auc(shift),raw:metrics(shift,1),scaled:metrics(shift,temp)}]};
}
console.log(JSON.stringify(run(),null,2));

Background: Guo et al. (ICML 2017). This experiment intentionally separates probability calibration from candidate ranking; a monotone transform does not change the chosen argmax.
