Held-out gate selection, noisy queries and reference baselines

Exploratory synthetic experiment; AI-assisted code and visualization. Six conditions (shared Gaussian component ρ ∈ {0, .35, .7}; overwritten fraction ∈ {0, .75}). Per condition: 32 validation and 64 independent test episodes; 48 keys, dimension 24, eight one-hot classes. Select the additive and delta gates on clean validation recall only; freeze gates for all three test query-noise levels. Query q = normalize(k + σ ε), ε ~ N(0,I/24). Same test episodes and perturbations are paired across methods. 576 independent episodes across both splits, 2,880 validation gate evaluations, 4,608 test method/noise evaluations. No model training or GPU timing.

Offline ridge fits all latest labels using λ=.1, hence has access to information not retained by the online state. Explicit retrieval keeps all latest key-label pairs and uses nearest cosine similarity; neither is a matched-state-budget competitor. Clean explicit retrieval is a sanity check. Error bars: one standard error across test episodes; delta minus additive intervals are mean ± 2 paired SE (approximate, unadjusted).

Reproduce every result (JavaScript, Node.js)

Save the following code as study.js and run node study.js. No dependencies; deterministic seeded generator; output is the full unrounded JSON result table.

function study(){
 const D=24,M=48,C=8,NV=32,NT=64,noise=[0,.35,.7],rs=[0,.35,.7],fs=[0,.75];
 function random(seed){let s=seed>>>0;return ()=>{s=(s+0x6D2B79F5)>>>0;let z=s;z=Math.imul(z^(z>>>15),z|1);z^=z+Math.imul(z^(z>>>7),z|61);return ((z^(z>>>14))>>>0)/4294967296;};}
 function normal(r){return Math.sqrt(-2*Math.log(Math.max(r(),1e-15)))*Math.cos(2*Math.PI*r());}
 function unit(a){const norm=Math.sqrt(a.reduce((s,v)=>s+v*v,0));return a.map(v=>v/norm);}
 const dot=(a,b)=>a.reduce((s,v,i)=>s+v*b[i],0);
 const argmax=a=>a.indexOf(Math.max(...a));
 const configs=[...[.9,.98,1].map(a=>({name:`A α=${a}`,a,b:1,delta:false})),...[.9,.98,1].flatMap(a=>[.25,.5,1,1.5].map(b=>({name:`D α=${a}, β=${b}`,a,b,delta:true})))];
 function episode(rho,f,seed){const r=random(seed),u=Array.from({length:D},()=>normal(r));const k=Array.from({length:M},()=>unit(Array.from({length:D},(_,j)=>Math.sqrt(rho)*u[j]+Math.sqrt(1-rho)*normal(r))));const y=Array.from({length:M},()=>Math.floor(r()*C)),latest=y.slice(),idx=Array.from({length:M},(_,i)=>i);for(let i=M-1;i>0;i--){let j=Math.floor(r()*(i+1));[idx[i],idx[j]]=[idx[j],idx[i]]}const events=y.map((v,i)=>[i,v]);for(const i of idx.slice(0,Math.round(f*M))){latest[i]=(y[i]+1+Math.floor(r()*(C-1)))%C;events.push([i,latest[i]])}const overwritten=new Set(idx.slice(0,Math.round(f*M)));const eps=k.map(()=>Array.from({length:D},()=>normal(r)/Math.sqrt(D)));const qs=noise.map(z=>k.map((v,i)=>unit(v.map((x,j)=>x+z*eps[i][j]))));return {k,latest,events,qs,overwritten};}
 function memory(ep,c){const S=Array.from({length:C},()=>Array(D).fill(0));for(const [i,v] of ep.events){const k=ep.k[i];for(let j=0;j<C;j++){const row=S[j];for(let d=0;d<D;d++)row[d]*=c.a;const err=(j===v?1:0)-(c.delta?dot(row,k):0);for(let d=0;d<D;d++)row[d]+=c.b*err*k[d];}}return q=>argmax(S.map(row=>dot(row,q)));}
 function ridge(ep){const A=Array.from({length:D},(_,i)=>Array.from({length:D},(_,j)=>i===j?.1:0)),B=Array.from({length:D},()=>Array(C).fill(0));for(let t=0;t<M;t++){const k=ep.k[t];for(let i=0;i<D;i++){B[i][ep.latest[t]]+=k[i];for(let j=0;j<D;j++)A[i][j]+=k[i]*k[j]}}for(let i=0;i<D;i++){const z=A[i][i];if(!(z>0))throw Error("Nonpositive ridge pivot");for(let j=i;j<D;j++)A[i][j]/=z;for(let j=0;j<C;j++)B[i][j]/=z;for(let p=0;p<D;p++){if(p===i)continue;const z2=A[p][i];for(let j=i;j<D;j++)A[p][j]-=z2*A[i][j];for(let j=0;j<C;j++)B[p][j]-=z2*B[i][j];}}return q=>argmax(Array.from({length:C},(_,j)=>q.reduce((sum,x,i)=>sum+x*B[i][j],0)));}
 function metrics(ep,predict,q){let all=0,ow=0,unch=0;for(let i=0;i<M;i++){const hit=+(predict(q[i])===ep.latest[i]);all+=hit;if(ep.overwritten.has(i))ow+=hit;else unch+=hit;}return [all/M,ep.overwritten.size?ow/ep.overwritten.size:null,unch/(M-ep.overwritten.size)];}
 function mean(a){return a.reduce((s,x)=>s+x,0)/a.length;}
 function se(a){const m=mean(a);return Math.sqrt(a.reduce((s,x)=>s+(x-m)**2,0)/(a.length-1)/a.length);}
 const results=[];for(let ri=0;ri<rs.length;ri++)for(let fi=0;fi<fs.length;fi++){const rho=rs[ri],f=fs[fi],base=721000+ri*10000+fi*1000;const val=Array.from({length:NV},(_,i)=>episode(rho,f,base+i));const valScores=configs.map(c=>mean(val.map(e=>metrics(e,memory(e,c),e.qs[0])[0])));const best=(start,end)=>Array.from({length:end-start},(_,i)=>i+start).reduce((a,b)=>valScores[b]>valScores[a]?b:a);const ai=best(0,3),di=best(3,configs.length);const test=Array.from({length:NT},(_,i)=>episode(rho,f,base+500+i));const rows=noise.map((z,ni)=>{const es=test.map(e=>{const predictors=[memory(e,configs[ai]),memory(e,configs[di]),ridge(e),q=>e.latest[argmax(e.k.map(k=>dot(k,q)))]];return predictors.map(p=>metrics(e,p,e.qs[ni]));});const methods=Array.from({length:4},(_,j)=>{const a=es.map(e=>e[j][0]);return {mean:mean(a),se:se(a),overwritten:f?mean(es.map(e=>e[j][1])):null,untouched:mean(es.map(e=>e[j][2]))};});const diffs=es.map(e=>e[1][0]-e[0][0]);return {noise:z,methods,diff:mean(diffs),half:2*se(diffs)};});if(Math.abs(rows[0].methods[3].mean-1)>1e-12)throw Error("Exact query nearest key sanity failure");results.push({rho,f,a:configs[ai].name,d:configs[di].name,va:valScores[ai],vd:valScores[di],rows});}return results;
}
